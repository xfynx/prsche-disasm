cmake_minimum_required(VERSION 3.20)
project(porsche_recovered_links LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Original x86 MSVC ABI required")
endif()
add_library(porsche_recovered_links STATIC ../../source/platform/recovered_links.cpp)
target_include_directories(porsche_recovered_links PRIVATE ../../source/include)
target_compile_features(porsche_recovered_links PRIVATE cxx_std_17)
target_compile_options(porsche_recovered_links PRIVATE /W4 /EHsc /permissive-)
